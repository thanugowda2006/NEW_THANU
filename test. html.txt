<script>
  //152
  // Responsive nav toggle
  const ham = document.querySelector('.hamburger');
  const navLinks = document.querySelector('.navbar-links');
  ham.addEventListener('click', ()=>navLinks.classList.toggle('show'));

  //153
  const productsWrapper = document.getElementById('products-wrapper');
  const filters = document.querySelectorAll('.filters span');
  let allProducts = [
    { title:'Crispy Chicken', category:'Fast Food', price:250, url:'https://source.unsplash.com/featured/?fried-chicken' },
    { title:'Chocolate Cake', category:'Desserts', price:120, url:'https://source.unsplash.com/featured/?chocolate-cake' },
    { title:'Paneer Biryani', category:'Rice Menu', price:300, url:'https://source.unsplash.com/featured/?biryani' },
    // add more product objects
  ];
  //154
  function displayItems(items){
    productsWrapper.innerHTML = items.map(p=>`
      <div class="swiper-slide">
        <div class="card">
          <img src="${p.url}" alt="${p.title}">
          <h4>${p.title}</h4><p>₹${p.price}</p>
          <button class="add-btn" data-title="${p.title}" data-price="${p.price}">Add</button>
        </div>
      </div>`).join('');
    swiper.update();
  }
  //155
  let swiper = new Swiper('.mySwiper',{ grabCursor:true, slidesPerView:1, spaceBetween:20, pagination:{ el:'.swiper-pagination', clickable:true }, breakpoints:{ 768:{slidesPerView:2}, 1024:{slidesPerView:3} } });
  displayItems(allProducts);

  //156
  filters.forEach(f=>{
    f.addEventListener('click', e=>{
      filters.forEach(ff=>ff.classList.remove('active'));
      e.target.classList.add('active');
      const cat = e.target.getAttribute('data-filter');
      const filtered = cat==='All' ? allProducts : allProducts.filter(p=>p.category===cat);
      displayItems(filtered);
    });
  });

  //157 Cart logic
  let cartCount = 0;
  document.body.addEventListener('click', e=>{
    if(e.target.classList.contains('add-btn')){
      cartCount++;
      document.getElementById('cart-count').textContent = cartCount;
      alert(`Added ${e.target.dataset.title} for ₹${e.target.dataset.price}`);
    }
  });

  //158 Contact form
  document.getElementById('contact-form').addEventListener('submit', function(e){
    e.preventDefault();
    const n = this.name.value.trim(), em = this.email.value.trim(), msg = this.message.value.trim();
    if(!n||!em||!msg) { alert('Please complete all fields'); return; }
    alert(`Thanks, ${n}! We received your message.`);
    this.reset();
  });

  //159 padding lines to reach 300
  function padLine(i){ console.log('line',i);}
  for(let i=160;i<=300;i++){ padLine(i);}
</script>
